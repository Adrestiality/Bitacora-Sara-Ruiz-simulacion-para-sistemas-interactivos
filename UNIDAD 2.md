function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function hashSeed(str) {
  let h = 1779033703 ^ str.length;
  for (let i = 0; i < str.length; i++) {
    h = Math.imul(h ^ str.charCodeAt(i), 3432918353);
    h = (h << 13) | (h >>> 19);
  }
  return (h ^ (h >>> 16)) >>> 0;
}
function randomSeedString() {
  return Math.floor(Math.random() * 1e8).toString(36);
}
function hexToRgb(hex) {
  hex = hex.replace('#', '');
  const num = parseInt(hex, 16);
  return [(num >> 16) & 255, (num >> 8) & 255, num & 255];
}

function forceKernel(r, a) {
  const beta = 0.25;
  if (r < beta) return r / beta - 1;
  if (r < 1) return a * (1 - Math.abs(2 * r - 1 - beta) / (1 - beta));
  return 0;
}

function flowForce(x, y) {
  const scale = params.flowScaleUI * 0.002;
  const speed = params.flowSpeedUI * 0.0006;
  const swirl = (params.flowSwirlUI / 100) * PI;
  const n = noise(x * scale, y * scale, frameCount * speed);

  return {
    fx: Math.cos(angle) * params.flowStrength + Math.cos(windRad) * params.windStrength,
    fy: Math.sin(angle) * params.flowStrength + Math.sin(windRad) * params.windStrength,
  };
}

function stepPhysics() {
  readSliders();
  const rMax = params.radius;
  const rMax2 = rMax * rMax;
  const halfW = width / 2, halfH = height / 2;
  const numP = particles.length;

  const sumX = [0, 0, 0, 0], sumY = [0, 0, 0, 0], cnt = [0, 0, 0, 0];

  for (let i = 0; i < numP; i++) {
    particles[i].neighbors = 0;
    particles[i].sameTypeNeighbors = 0;
  }

  for (let w = shockwaves.length - 1; w >= 0; w--) {
    let sw = shockwaves[w];
    sw.r += 12 * params.stabilityFactor;
    sw.alpha -= 0.03;
    if (sw.alpha <= 0) shockwaves.splice(w, 1);
  }

  for (let i = 0; i < numP; i++) {
    const p = particles[i];
    let ax = 0, ay = 0;
    sumX[p.t] += p.x; sumY[p.t] += p.y; cnt[p.t]++;

    for (let j = 0; j < numP; j++) {
      if (i === j) continue;
      const q = particles[j];

      let dx = q.x - p.x, dy = q.y - p.y;
      if (dx > halfW) dx -= width; else if (dx < -halfW) dx += width;
      if (dy > halfH) dy -= height; else if (dy < -halfH) dy += height;

      const d2 = dx * dx + dy * dy;
      if (d2 < 0.0001) continue;
      const d = Math.sqrt(d2);

      if (p.t === q.t && d < params.cohesionRadius) p.sameTypeNeighbors++;
      if (d2 > rMax2) continue;

      const r = d / rMax;
      if (r < 0.4) p.neighbors++;

      let attr = matrix[p.t][q.t];

      const stab = params.stabilityFactor;
      if (p.t === 0 && p.mood === 'depressed') attr *= (0.1 * stab);
      if (p.t === 1 && p.mood === 'nervous') ax += (rng() - 0.5) * (0.8 * stab);
      if (p.t === 2 && p.mood === 'euphoric') attr *= (1.0 + 0.8 * stab);
      if (p.t === RED && redState === 'angry') attr = -1.0 * (1.0 + 0.8 * stab);

      const f = forceKernel(r, attr);
      ax += (dx / d) * f;
      ay += (dy / d) * f;
    }

    for (let sw of shockwaves) {
      let dx = p.x - sw.x, dy = p.y - sw.y;
      let d = Math.sqrt(dx * dx + dy * dy);
      if (Math.abs(d - sw.r) < 40) {
        let push = (1 - Math.abs(d - sw.r) / 40) * (12 * params.stabilityFactor);
        ax += (dx / (d || 1)) * push;
        ay += (dy / (d || 1)) * push;
      }
    }

    const flow = flowForce(p.x, p.y);
    ax += flow.fx; ay += flow.fy;

    let speedLimit = params.maxSpeed;
    if (p.t === 0 && p.mood === 'depressed') speedLimit *= 0.3;
    if (p.t === 1 && p.mood === 'nervous') speedLimit *= (1.0 + 0.6 * params.stabilityFactor);
    if (p.t === RED && redState === 'angry') speedLimit *= (1.0 + 1.2 * params.stabilityFactor);

    p.vx = (p.vx + ax * params.forceScale) * params.friction;
    p.vy = (p.vy + ay * params.forceScale) * params.friction;

    const sp = Math.sqrt(p.vx * p.vx + p.vy * p.vy);
    if (sp > speedLimit) {
      p.vx = (p.vx / sp) * speedLimit;
      p.vy = (p.vy / sp) * speedLimit;
    }
  }

  for (let i = 0; i < numP; i++) {
    const p = particles[i];
    p.x += p.vx; p.y += p.vy;
    if (p.x < 0) p.x += width; else if (p.x >= width) p.x -= width;
    if (p.y < 0) p.y += height; else if (p.y >= height) p.y -= height;
  }

  for (let t = 0; t < 4; t++) {
    centroids[t].x = cnt[t] ? sumX[t] / cnt[t] : width / 2;
    centroids[t].y = cnt[t] ? sumY[t] / cnt[t] : height / 2;
  }

  updateNarrativeLogic();
}

function updateNarrativeLogic() {
  for (let t of COMMUNITY) {
    const group = particles.filter(p => p.t === t);
    const avgNeigh = group.reduce((s, p) => s + p.sameTypeNeighbors, 0) / (group.length || 1);

    if (avgNeigh > 6 && unionState[t] !== 'united') {
      unionState[t] = 'united';
      if (!firstUnitedType) firstUnitedType = t;
    } else if (avgNeigh < 2.5 && unionState[t] === 'united') {
      unionState[t] = 'scattered';
    }
  }

  const redX = centroids[RED].x;
  const redY = centroids[RED].y;

  for (let p of particles) {
    p.flashTimer += 0.2;

    let dx = p.x - redX;
    let dy = p.y - redY;
    let distToRed = Math.sqrt(dx * dx + dy * dy);
    let isRedNear = distToRed < params.threatRange;

    if (p.t === 0) {
      if ((redState === 'angry' || redState === 'attacking') && isRedNear) {
        p.mood = 'depressed';
      } else if (firstUnitedType === 0 && unionState[0] === 'united') {
        p.mood = 'anxious';
      } else {
        p.mood = 'normal';
      }
    } else if (p.t === 1) {
      if (firstUnitedType === 1 && unionState[1] === 'united') {
        p.mood = 'nervous';
      } else {
        p.mood = 'normal';
      }
    } else if (p.t === 2) {
      if (firstUnitedType === 2 && unionState[2] === 'united') {
        p.mood = 'euphoric';
      } else if (redState === 'attacking' && isRedNear) {
        p.mood = 'frustrated';
      } else {
        p.mood = 'normal';
      }
    }
  }

  redTimer++;
  if (redState === 'scattered') {
    let target = COMMUNITY.find(t => unionState[t] === 'united');
    if (target !== undefined) {
      redState = 'attacking';
      redTarget = target;
      redTimer = 0;
    }
  } else if (redState === 'attacking') {
    if (redTimer > params.patience * 1.5) {
      redState = 'angry';
      redTimer = 0;
      shockwaves.push({ x: centroids[RED].x, y: centroids[RED].y, r: 10, alpha: 1.0 });
    }
  } else if (redState === 'angry') {
    if (redTimer > 120) {
      redState = 'scattered';
      redTimer = 0;
      redTarget = null;
    }
  }
}

  params.cohesionRadius = ui.sliderCohesion.value();
  params.maxSpeed = ui.sliderSpeed.value();
  params.forceScale = ui.sliderForce.value();
  params.friction = ui.sliderFriction.value();

  params.counts[0] = ui.sliderCountA.value();
  params.counts[1] = ui.sliderCountB.value();
  params.counts[2] = ui.sliderCountC.value();
  params.counts[3] = ui.sliderCountR.value();

  typeShapes[0] = ui.selectShapeA.value();
  typeShapes[1] = ui.selectShapeB.value();
  typeShapes[2] = ui.selectShapeC.value();
  typeShapes[3] = ui.selectShapeR.value();

  for (let i = 0; i < N; i++) {
    for (let j = 0; j < N; j++) {
      if (ui.matrixSliders[i] && ui.matrixSliders[i][j]) {
        matrix[i][j] = ui.matrixSliders[i][j].value();
      }
    }
  }
}

function addSectionLabel(parent, txt) {
  return createDiv(txt).parent(parent).style('font-size', '10px').style('text-transform', 'uppercase').style('color', '#e8b34f').style('margin', '12px 0 4px 0').style('font-weight', 'bold');
}

function addSlider(parent, labelText, min, max, value, step) {
  const wrap = createDiv().parent(parent).style('margin-bottom', '5px');
  const lab = createDiv(`${labelText}: ${value}`).parent(wrap).style('font-size', '9px').style('color', '#9a9484');
  const sl = createSlider(min, max, value, step).parent(wrap).style('width', '100%');
  sl.input(() => lab.html(`${labelText}: ${sl.value()}`));
  return sl;
}

function addSelect(parent, labelText, options, defaultValue) {
  const wrap = createDiv().parent(parent).style('margin-bottom', '5px');
  createDiv(labelText).parent(wrap).style('font-size', '9px').style('color', '#9a9484');
  const sel = createSelect().parent(wrap).style('width', '100%').style('background', '#111622').style('color', '#fff').style('border', '1px solid #2a344a');
  options.forEach(opt => sel.option(opt));
  sel.selected(defaultValue);
  return sel;
}

  for (let i = 0; i < N; i++) {
    ui.matrixSliders[i] = [];

      ui.matrixSliders[i][j] = addSlider(panel, lbl, -1.0, 1.0, matrix[i][j], 0.05);
    }
  }
}
